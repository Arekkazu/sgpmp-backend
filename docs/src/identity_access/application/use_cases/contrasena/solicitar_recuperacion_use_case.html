src.identity_access.application.use_cases.contrasena.solicitar_recuperacion_use_case

Caso de uso: solicitud de recuperación de contraseña.

Aplica rate limiting por IP (máx 3 por hora), genera un token de recuperación y envía el correo correspondiente. Retorna siempre un mensaje genérico para evitar enumeración de usuarios registrados.

 23TIPO_SOLICITUD_RECUPERACION = 7

 25_MENSAJE_GENERICO = "Si el correo está registrado, recibirás instrucciones para recuperar tu contraseña en unos minutos."

 29    """Orquesta el inicio del flujo de recuperación de contraseña."""

 32        self,
 33        usuarios_repo: UsuarioRepository,
 34        cuentas_repo: CuentaRepository,
 35        eventos_repo: EventoRepository,
 36        db: Session,
 37        notificacion_service=None,
 38    ):
 39        """Inicializa el use case.
 40
 41        Args:
 42            usuarios_repo: Repositorio de dominio del agregado Usuario.
 43            cuentas_repo: Repositorio de dominio del agregado Cuenta (token de recuperación).
 44            eventos_repo: Repositorio de dominio de eventos (rate limiting y auditoría).
 45            db: Sesión SQLAlchemy activa del request.
 46            notificacion_service: Servicio de notificaciones opcional.
 47        """
 48        self.usuarios_repo = usuarios_repo
 49        self.cuentas_repo = cuentas_repo
 50        self.eventos_repo = eventos_repo
 51        self.db = db
 52        self.notificacion_service = notificacion_service
 53
 54    def execute(self, dto: SolicitarRecuperacionDTO, ip: str) -> str:
 55        """Inicia el proceso de recuperación de contraseña para el correo indicado.
 56
 57        Si la cuenta está en estado PENDIENTE, redirige al flujo de activación
 58        en lugar del de recuperación. Si el correo no existe o la cuenta está
 59        eliminada, retorna el mensaje genérico sin revelar información.
 60
 61        Args:
 62            dto: Correo electrónico del usuario que solicita la recuperación.
 63            ip: IP del cliente, usada para el rate limiting por hora.
 64
 65        Returns:
 66            Mensaje genérico que no revela si el correo está registrado.
 67
 68        Raises:
 69            BusinessRuleError: Si se supera el límite de 3 solicitudes por hora
 70                desde la misma IP. HTTP 422.
 71        """
 72        # 1. Rate limit por IP: máx 3 solicitudes por hora
 73        ahora = datetime.now(timezone.utc)
 74        hace_una_hora = ahora - timedelta(hours=1)
 75        solicitudes = self.eventos_repo.contar_solicitudes_recuperacion_por_ip(ip, hace_una_hora)
 76        if solicitudes >= MAX_SOLICITUDES_POR_HORA:
 77            proxima_vez = hace_una_hora + timedelta(hours=1)
 78            raise BusinessRuleError(
 79                code="LIMITE_SOLICITUDES_EXCEDIDO",
 80                message=(
 81                    f"Límite de solicitudes excedido para su conexión. Por seguridad, solo se "
 82                    f"permiten {MAX_SOLICITUDES_POR_HORA} intentos de recuperación por hora. "
 83                    f"Podrá intentarlo de nuevo a las {proxima_vez.strftime('%H:%M:%S')}."
 84                ),
 85            )
 86
 87        # 2. Buscar usuario — flujo interno solo si existe y no está eliminado
 88        correo = str(dto.correo_electronico)
 89        usuario = self.usuarios_repo.obtener_por_correo(Email(correo))
 90        if usuario is None:
 91            return _MENSAJE_GENERICO
 92
 93        cuenta = self.cuentas_repo.obtener_por_usuario(usuario.id_usuario)
 94        if cuenta is None or cuenta.id_estado_cuenta == Cuenta.ESTADO_ELIMINADO:
 95            return _MENSAJE_GENERICO
 96
 97        # 3. Cuenta en PENDIENTE: enviar email de activación en su lugar
 98        if cuenta.esta_pendiente():
 99            token_activacion = cuenta.token_activacion_actual
100            if token_activacion:
101                try:
102                    self.eventos_repo.registrar(
103                        tipo_evento=TIPO_SOLICITUD_RECUPERACION,
104                        exitoso=True,
105                        id_usuario=usuario.id_usuario,
106                        detalle={"ip": ip, "motivo": "cuenta_pendiente_redirigido_a_activacion"},
107                    )
108                    self.db.commit()
109                except Exception:
110                    self.db.rollback()
111                    raise
112                send_email(
113                    to=correo,
114                    subject="Activa tu cuenta en SGPMP",
115                    html_body=activation_email(usuario.nombre, token_activacion),
116                )
117            return _MENSAJE_GENERICO
118
119        # 4. Generar token de recuperación y guardarlo
120        token = secrets.token_urlsafe(32)
121        try:
122            cuenta.asignar_token_recuperacion(token, ahora)
123            self.cuentas_repo.guardar(cuenta)
124            self.eventos_repo.registrar(
125                tipo_evento=TIPO_SOLICITUD_RECUPERACION,
126                exitoso=True,
127                id_usuario=usuario.id_usuario,
128                detalle={"ip": ip, "motivo": "token_generado"},
129            )
130            self.db.commit()
131        except Exception:
132            self.db.rollback()
133            raise
134
135        # 5. Enviar email de recuperación (post-commit)
136        send_email(
137            to=correo,
138            subject="Restablece tu contraseña en SGPMP",
139            html_body=recovery_email(usuario.nombre, token),
140        )
141
142        if self.notificacion_service:
143            self.notificacion_service.notificar(
144                tipo_evento=TIPO_SOLICITUD_RECUPERACION,
145                id_usuario=usuario.id_usuario,
146                correo_destino=correo,
147            )
148
149        return _MENSAJE_GENERICO
