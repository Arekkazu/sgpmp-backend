src.identity_access.application.use_cases.cuentas.gestionar_cuenta_use_case

Caso de uso: gestión administrativa del estado de una cuenta de usuario.

Permite a un administrador activar, inactivar, bloquear o eliminar la cuenta de otro usuario. Protege al último administrador activo y valida las transiciones de estado permitidas.

 42class GestionarCuentaUseCase:
 43    """Orquesta el cambio de estado de la cuenta de un usuario por un administrador."""
 44
 45    def __init__(

 55
 56        Args:

 59            eventos_repo: Repositorio de dominio de eventos (registro de auditoría).
 60            sesiones_repo: Repositorio de dominio de sesiones (invalidación).
 61            db: Sesión SQLAlchemy activa del request.
 62            notificacion_service: Servicio de notificaciones opcional.
 63        """
 64        self.usuarios_repo = usuarios_repo
 65        self.cuentas_repo = cuentas_repo
 66        self.eventos_repo = eventos_repo
 67        self.sesiones_repo = sesiones_repo
 68        self.db = db
 69        self.notificacion_service = notificacion_service

 71    def execute(self, id_usuario: int, dto: GestionarCuentaDTO, usuario_actual: UsuarioActual) -> None:
 72        """Aplica la acción de gestión sobre la cuenta del usuario indicado.
 73
 74        Args:
 75            id_usuario: ID del usuario cuya cuenta se va a gestionar.
 76            dto: Datos de la acción (acción a ejecutar y motivo opcional).
 77            usuario_actual: Usuario administrador que ejecuta la acción.
 78
 79        Raises:
 80            AuthorizationError: Si el actor no es administrador o intenta
 81                gestionar su propia cuenta. HTTP 403.
 82            NotFoundError: Si el usuario o su cuenta no existen. HTTP 404.
 83            ValidationError: Si la acción crítica no tiene motivo, o la cuenta
 84                ya está en el estado solicitado. HTTP 400.
 85            BusinessRuleError: Si la transición de estado no es válida o se
 86                intenta desactivar al único administrador activo. HTTP 422.
 87        """
 88        # 1. Solo administradores
 89        if usuario_actual.id_rol != ROL_ADMINISTRADOR:
 90            raise AuthorizationError(
 91                code="ACCESO_DENEGADO",
 92                message=(
 93                    "Acceso denegado. Se requieren privilegios administrativos para gestionar "
 94                    "estados de cuenta. Este intento de acceso ha sido registrado en el log de auditoría."
 95                ),
 96            )
 97
 98        # 2. Admin no puede gestionar su propia cuenta
 99        if id_usuario == usuario_actual.id_usuario:
100            raise AuthorizationError(
101                code="AUTOEDICION_ESTADO_PROHIBIDA",
102                message=(
103                    "Operación no permitida. Un administrador no puede gestionar el estado "
104                    "de su propia cuenta."
105                ),
106            )
107
108        # 3. Buscar usuario y cuenta
109        usuario = self.usuarios_repo.obtener_por_id(id_usuario)
110        if usuario is None:
111            raise NotFoundError(
112                code="USUARIO_NO_ENCONTRADO",
113                message=f"Error de referencia. El usuario con ID {id_usuario} no existe en los registros actuales del sistema.",
114            )
115
116        cuenta = self.cuentas_repo.obtener_por_usuario(id_usuario)
117        if cuenta is None:
118            raise NotFoundError(
119                code="CUENTA_NO_ENCONTRADA",
120                message=f"El usuario con ID {id_usuario} no tiene una cuenta asociada.",
121            )
122
123        # 4. Motivo obligatorio para acciones críticas
124        motivo = dto.motivo_accion
125        if dto.accion_cuenta in ACCIONES_CRITICAS and not motivo:
126            raise ValidationError(
127                code="MOTIVO_REQUERIDO",
128                message=(
129                    "Campo obligatorio faltante. Para las acciones de Desactivación o Eliminación, "
130                    "es obligatorio registrar una justificación en el campo 'motivo_accion' para fines de auditoría."
131                ),
132                field="motivo_accion",
133            )
134        motivo = motivo or _MOTIVO_DEFAULT
135
136        # 5. Determinar estado destino y validar transición
137        nuevo_estado = ACCION_A_ESTADO[dto.accion_cuenta]
138        estado_actual = cuenta.id_estado_cuenta
139
140        if nuevo_estado == estado_actual:
141            raise ValidationError(
142                code="ESTADO_SIN_CAMBIO",
143                message=f"Inconsistencia de estado. La cuenta ya se encuentra en el estado solicitado.",
144            )
145
146        if estado_actual == 5:
147            raise BusinessRuleError(
148                code="TRANSICION_INVALIDA",
149                message=(
150                    "Transición de estado inválida. Una cuenta en estado 'ELIMINADO' tiene carácter "
151                    f"irreversible para preservar la integridad histórica. No se permite el cambio a estado {nuevo_estado}."
152                ),
153            )
154
155        if nuevo_estado not in TRANSICIONES_VALIDAS.get(estado_actual, set()):
156            raise BusinessRuleError(
157                code="TRANSICION_INVALIDA",
158                message=f"Transición de estado inválida ({estado_actual} → {nuevo_estado}).",
159            )
160
161        # 6. Protección del último administrador activo
162        if usuario.id_rol == ROL_ADMINISTRADOR and dto.accion_cuenta in ACCIONES_QUE_REDUCEN_ADMINS:
163            if self.cuentas_repo.contar_admins_activos() <= 1:
164                raise BusinessRuleError(
165                    code="ULTIMO_ADMIN_PROTEGIDO",
166                    message=(
167                        f"Operación denegada por seguridad. El usuario {id_usuario} es el único "
168                        "administrador activo del sistema. Debe designar y activar un nuevo "
169                        "administrador antes de proceder con la desactivación o eliminación."
170                    ),
171                )
172
173        # 7. Revocar sesiones ANTES del flush del estado (trigger las desactiva después)
174        if nuevo_estado in ESTADOS_QUE_INVALIDAN_SESION:
175            self.sesiones_repo.invalidar_todas_sesiones(cuenta.id_cuenta_usuario)
176
177        try:
178            # 8. Actualizar estado vía la entidad
179            cuenta.cambiar_estado(nuevo_estado, motivo)
180            self.cuentas_repo.guardar(cuenta)
181
182            # 9. Registrar en gestiones_cuenta
183            self.cuentas_repo.registrar_gestion(
184                id_cuenta_usuario=cuenta.id_cuenta_usuario,
185                accion=dto.accion_cuenta.value,
186                motivo=motivo,
187                id_responsable=usuario_actual.id_usuario,
188            )
189
190            # 10. Registrar evento de auditoría
191            self.eventos_repo.registrar(
192                tipo_evento=TIPO_CAMBIO_ESTADO,
193                exitoso=True,
194                id_usuario=usuario_actual.id_usuario,
195                detalle={
196                    "id_usuario_afectado": id_usuario,
197                    "accion": dto.accion_cuenta.value,
198                    "estado_anterior": estado_actual,
199                    "estado_nuevo": nuevo_estado,
200                    "motivo": motivo,
201                },
202            )
203
204            self.db.commit()
205        except Exception:
206            self.db.rollback()
207            raise
208
209        if self.notificacion_service:
210            self.notificacion_service.notificar(
211                tipo_evento=TIPO_CAMBIO_ESTADO,
212                id_usuario=id_usuario,
213                correo_destino=str(usuario.correo),
214            )

144                message=f"Inconsistencia de estado. La cuenta ya se encuentra en el estado solicitado.",
145            )
146
147        if estado_actual == 5:
148            raise BusinessRuleError(
149                code="TRANSICION_INVALIDA",
150                message=(
151                    "Transición de estado inválida. Una cuenta en estado 'ELIMINADO' tiene carácter "
152                    f"irreversible para preservar la integridad histórica. No se permite el cambio a estado {nuevo_estado}."
153                ),
154            )
155
156        if nuevo_estado not in TRANSICIONES_VALIDAS.get(estado_actual, set()):
157            raise BusinessRuleError(
158                code="TRANSICION_INVALIDA",
159                message=f"Transición de estado inválida ({estado_actual} → {nuevo_estado}).",
160            )
161
162        # 6. Protección del último administrador activo
163        if usuario.id_rol == ROL_ADMINISTRADOR and dto.accion_cuenta in ACCIONES_QUE_REDUCEN_ADMINS:
164            if self.cuentas_repo.contar_admins_activos() <= 1:
165                raise BusinessRuleError(
166                    code="ULTIMO_ADMIN_PROTEGIDO",
167                    message=(
168                        f"Operación denegada por seguridad. El usuario {id_usuario} es el único "
169                        "administrador activo del sistema. Debe designar y activar un nuevo "
170                        "administrador antes de proceder con la desactivación o eliminación."
171                    ),
172                )
173
174        # 7. Revocar sesiones ANTES del flush del estado (trigger las desactiva después)
175        if nuevo_estado in ESTADOS_QUE_INVALIDAN_SESION:
176            self.sesiones_repo.invalidar_todas_sesiones(cuenta.id_cuenta_usuario)
177
178        try:
179            # 8. Actualizar estado vía la entidad
180            cuenta.cambiar_estado(nuevo_estado, motivo)
181            self.cuentas_repo.guardar(cuenta)
182
183            # 9. Registrar en gestiones_cuenta
184            self.cuentas_repo.registrar_gestion(
185                id_cuenta_usuario=cuenta.id_cuenta_usuario,
186                accion=dto.accion_cuenta.value,
187                motivo=motivo,
188                id_responsable=usuario_actual.id_usuario,
189            )
190
191            # 10. Registrar evento de auditoría
192            self.eventos_repo.registrar(
193                tipo_evento=TIPO_CAMBIO_ESTADO,
194                exitoso=True,
195                id_usuario=usuario_actual.id_usuario,
196                detalle={
197                    "id_usuario_afectado": id_usuario,
198                    "accion": dto.accion_cuenta.value,
199                    "estado_anterior": estado_actual,
200                    "estado_nuevo": nuevo_estado,
201                    "motivo": motivo,
202                },
203            )
204
205            self.db.commit()
206        except Exception:
207            self.db.rollback()
208            raise
209
210        if self.notificacion_service:
211            self.notificacion_service.notificar(
212                tipo_evento=TIPO_CAMBIO_ESTADO,
213                id_usuario=id_usuario,
214                correo_destino=str(usuario.correo),
215            )

Aplica la acción de gestión sobre la cuenta del usuario indicado.

Args: id_usuario: ID del usuario cuya cuenta se va a gestionar. dto: Datos de la acción (acción a ejecutar y motivo opcional). usuario_actual: Usuario administrador que ejecuta la acción.

Raises: AuthorizationError: Si el actor no es administrador o intenta gestionar su propia cuenta. HTTP 403. NotFoundError: Si el usuario o su cuenta no existen. HTTP 404. ValidationError: Si la acción crítica no tiene motivo, o la cuenta ya está en el estado solicitado. HTTP 400. BusinessRuleError: Si la transición de estado no es válida o se intenta desactivar al único administrador activo. HTTP 422.
