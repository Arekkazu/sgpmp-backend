- Explicación de la contradicción ACTIVO vs 409 "estado CERRADO":
  Al invocar POST /activos-biologicos/9/fases con id_ciclo_productiva: 3, el backend responde HTTP 409:
  {"error_code":"ACTIVO_NO_OPERATIVO","message":"No se puede cambiar fase en activo ID 9 (estado CERRADO)."}
  Causa técnica: En base de datos, el trigger modulo2.trg_fn_fase_activo_estado_valido() consulta prioritariamente modulo2.historial_estados_activos_biologicos (ORDER BY fecha_cambio DESC LIMIT 1). Al encontrar CERRADO en el historial, el trigger lanza la excepción personalizada P0228 (INVALID_STATE), la cual es traducida por db_error_translator.py como HTTP 409 ACTIVO_NO_OPERATIVO.
- Respuesta de POST /suministros/medicamentos sobre Activo 9:
  Al intentar registrar medicamento sobre el Activo 9, el pipeline valida:
  Paso 1: Existencia (OK).
  Paso 2: id_estado in (1, 3) (OK, id_estado es 1).
  Paso 3: Fases activas > 0 (FALLA: Activo 9 no tiene fase activa).
  Respuesta real observada: HTTP 422 CICLO_NO_ABIERTO. El flujo se detiene en Paso 3 y nunca alcanza la validación de cantidad poblacional (Paso 8).

"""Ejecución de T4 + C2 + C4 en UNA SOLA corrida sandbox.

from __future__ import annotations

import json

import time

from datetime import datetime, timezone, timedelta
from pathlib import Path

from dotenv import dotenv_values

_DIR_ACTUAL = Path(__file__).resolve().parent
_RAIZ_REPO = Path(__file__).resolve().parents[7]
if not (_RAIZ_REPO / ".env.test").exists():
    _RAIZ_REPO = Path.cwd()
_ENV_TEST = dotenv_values(_RAIZ_REPO / ".env.test")


def env(nombre: str) -> str:
    v = os.environ.get(nombre) or _ENV_TEST.get(nombre)
    if not v:
        raise RuntimeError(f"Falta variable de entorno: {nombre}")
    return v


BASE_URL = env("TEST_BASE_URL")
SANDBOX_SECRET = env("TEST_SANDBOX_SECRET")
EMAIL = env("TEST_ADMIN_EMAIL")
PASSWORD = env("TEST_ADMIN_PASSWORD")

TRAZAS_PATH = _DIR_ACTUAL / "trazas_c2_c4.jsonl"


def sanitizar_cuerpo(data: any) -> any:
    if isinstance(data, dict):
        san = {}
        for k, v in data.items():
            if k.lower() in ("contrasena", "password", "token", "authorization"):
                san[k] = "***REDACTED***"
            elif isinstance(v, (dict, list)):
                san[k] = sanitizar_cuerpo(v)
            else:
                san[k] = v
        return san
    elif isinstance(data, list):
        return [sanitizar_cuerpo(x) for x in data]
    return data


def log_traza(f_out, metodo: str, url: str, status_code: int, req_body: any, resp: requests.Response | None, duracion_ms: float):
    now_iso = datetime.now(timezone(timedelta(hours=-5))).isoformat()
    resp_body = None
    server_date = ""
    if resp is not None:
        server_date = resp.headers.get("Date", "")
        try:
            resp_body = resp.json()
        except Exception:
            resp_body = resp.text[:500]

    entry = {
        "timestamp_local": now_iso,
        "server_date": server_date,
        "duracion_ms": round(duracion_ms, 2),
        "metodo": metodo,
        "url": url,
        "status_code": status_code,
        "request_body": sanitizar_cuerpo(req_body),
        "response_body": sanitizar_cuerpo(resp_body),
    }
    f_out.write(json.dumps(entry, ensure_ascii=False) + "\n")
    f_out.flush()
    return entry


def payload_base(id_activo: int, dosis: float = 10.0, hora: str = "08:00") -> dict:

def main():
    run_id = f"tc-m05-g15-c2-c4-{uuid.uuid4()}"
    print(f"Iniciando corrida C2-C4 con run_id: {run_id}")

    session = requests.Session()
    headers_base = {
        "X-Test-Run-Id": run_id,
        "X-Sandbox-Secret": SANDBOX_SECRET,
        "X-Zona-Horaria": "America/Bogota",
    }
    session.headers.update(headers_base)

    resultados_resumen = []
    f_trazas = open(TRAZAS_PATH, "w", encoding="utf-8")

    try:
        # 1. Login único
        t0 = time.perf_counter()
        r_login = session.post(
            f"{BASE_URL}/sesiones/",
            json={"correo_electronico": EMAIL, "contrasena": PASSWORD},
            timeout=30,
        )
        ms = (time.perf_counter() - t0) * 1000
        log_traza(f_trazas, "POST", f"{BASE_URL}/sesiones/", r_login.status_code, {"correo_electronico": EMAIL, "contrasena": "***"}, r_login, ms)
        assert r_login.status_code == 200, f"Login fallo: {r_login.status_code} {r_login.text}"

def sanitizar(data: any) -> any:
    if isinstance(data, dict):
        san = {}
        for k, v in data.items():
            if k.lower() in ("contrasena", "password", "token", "authorization"):
                san[k] = "***REDACTED***"
            elif isinstance(v, (dict, list)):
                san[k] = sanitizar(v)
            else:
                san[k] = v
        return san
    elif isinstance(data, list):
        return [sanitizar(x) for x in data]
    return data


def log_traza(f_out, metodo: str, url: str, status_code: int, req_body: any, resp: requests.Response | None, duracion_ms: float):
    now_iso = datetime.now(timezone(timedelta(hours=-5))).isoformat()
    resp_body = None
    server_date = ""
    if resp is not None:
        server_date = resp.headers.get("Date", "")
        try:
            resp_body = resp.json()
        except Exception:
            resp_body = resp.text[:500]

    entry = {
        "timestamp_local": now_iso,
        "server_date": server_date,
        "duracion_ms": round(duracion_ms, 2),
        "metodo": metodo,
        "url": url,
        "status_code": status_code,
        "request_body": sanitizar(req_body),
        "response_body": sanitizar(resp_body),
    }
    f_out.write(json.dumps(entry, ensure_ascii=False) + "\n")
    f_out.flush()
    return entry


def hoy_str() -> str:
    return datetime.now(timezone(timedelta(hours=-5))).date().isoformat()
