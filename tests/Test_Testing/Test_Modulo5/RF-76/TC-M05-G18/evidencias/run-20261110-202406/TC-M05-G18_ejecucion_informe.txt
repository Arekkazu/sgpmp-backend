"""Genera resultado_TC-M05-G18.json a partir de junit_TC-M05-G18.xml y trazas_tc_m05_g18.json."""
from __future__ import annotations

import json
from pathlib import Path
import xml.etree.ElementTree as ET


def generar_resultado(run_dir: Path) -> dict:
    """Construye y valida el resumen oficial JSON a partir del JUnit XML y las trazas."""
    junit_path = run_dir / "junit_TC-M05-G18.xml"
    trazas_path = run_dir / "trazas_tc_m05_g18.json"
    out_path = run_dir / "resultado_TC-M05-G18.json"

    tree = ET.parse(junit_path)
    root = tree.getroot()
    testcases = root.findall(".//testcase")
    trazas = json.loads(trazas_path.read_text(encoding="utf-8"))

    if len(testcases) != len(trazas):
        raise ValueError(
            f"Incoherencia entre JUnit ({len(testcases)}) y trazas ({len(trazas)})."
        )

    checkpoints: list[dict[str, str]] = []
    ok_count = 0
    falla_count = 0

    for testcase, traza in zip(testcases, trazas):
        xml_fallo = testcase.find("failure") is not None or testcase.find("error") is not None
        xml_estado = "FALLA" if xml_fallo else "OK"
        traza_estado = traza["estado"]
        if xml_estado != traza_estado:
            raise ValueError(
                f"Estado divergente en {traza['test']}: XML={xml_estado} vs traza={traza_estado}"
            )

        if traza_estado == "OK":
            ok_count += 1
        else:
            falla_count += 1

        checkpoints.append(
            {
                "paso": (
                    f"{traza['test']} (plano {traza['caso_plano']} / orig {traza['caso_original']}) "
                    f"- {testcase.attrib['name']}"
                ),
                "esperado": traza["esperado"],
                "obtenido": traza["obtenido"],
                "estado": traza_estado,
            }
        )

    veredicto = "Aprobado" if falla_count == 0 and len(checkpoints) > 0 else "Rechazado"
    payload = {
        "tc": "TC-M05-G18",
        "summary": {
            "total": len(checkpoints),
            "ok": ok_count,
            "falla": falla_count,
            "veredicto": veredicto,
        },
        "checkpoints": checkpoints,
    }
    out_path.write_text(json.dumps(payload, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    return payload


if __name__ == "__main__":
    resultado = generar_resultado(Path(__file__).resolve().parent)
    print(json.dumps(resultado["summary"], ensure_ascii=False))

+def _relanzar_con_rollback(cuerpo_exc: BaseException, del_resumen: str) -> None:
+    """Preserva la excepcion original del cuerpo adjuntando el resultado del DELETE (C10)."""
+    if hasattr(cuerpo_exc, "add_note"):
+        cuerpo_exc.add_note(f"Rollback sandbox: {del_resumen}")
+    if isinstance(cuerpo_exc, AssertionError):
+        raise AssertionError(f"{cuerpo_exc} [rollback: {del_resumen}]") from cuerpo_exc
+    if getattr(cuerpo_exc, "args", None) and isinstance(cuerpo_exc.args[0], str):
+        cuerpo_exc.args = (f"{cuerpo_exc.args[0]} [rollback: {del_resumen}]", *cuerpo_exc.args[1:])
+    raise cuerpo_exc

+def _ejecutar_corrida_sandbox(
+    *,
+    base_url: str,
+    secret: str,
+    run_id: str,
+    cuerpo_fn: Callable[[requests.Session], Any],
+) -> Any:
+    """Ejecuta una corrida sandbox garantizando DELETE /test-control/runs/{run_id} seguro en finally (C10)."""
+    session = requests.Session()
+    session.headers.update(
+        {
+            "X-Test-Run-Id": run_id,
+            "X-Sandbox-Secret": secret,
+            "X-Zona-Horaria": "America/Bogota",
+        }
+    )
+    cuerpo_exc: Optional[BaseException] = None
+    resultado: Any = None
+    try:
+        resultado = cuerpo_fn(session)
+    except BaseException as exc:
+        cuerpo_exc = exc
+    finally:
+        del_status: Optional[int] = None
+        del_err: Optional[str] = None
+        try:
+            r_del = session.delete(f"{base_url}/test-control/runs/{run_id}", timeout=45)
+            del_status = r_del.status_code
+        except Exception as exc_del:
+            del_err = f"{type(exc_del).__name__}: {exc_del}"
+        finally:
+            session.close()
+
+        del_resumen = f"DELETE={del_status}" if del_err is None else f"DELETE_ERR={del_err}"
+        if cuerpo_exc is not None:
+            _relanzar_con_rollback(cuerpo_exc, del_resumen)
+        assert del_status == 200, f"Rollback corrida {run_id} fallo: {del_resumen}"
+    return resultado
