# Prueba de la herramienta del reintento 1 de TC-M08-G018 (no es parte de la corrida oficial).
# Emula los pasos del runner v4 que no dependen de PowerShell: mock en 127.0.0.1, Newman con los mismos reporteros,
# el mismo parseo de la consola (mismas expresiones que el .ps1) y el conversor convertir_json_g018.mjs.
# Uso: python3 -I emular_runner.py <coleccion> <mock_api.py> <conversor.mjs> <modo> <salida> <puerto>
import os, re, subprocess, sys, time

col, mock, conversor, modo, salida, puerto = sys.argv[1:7]
tc, suf = "TC-M08-G018", "_reintento1"
os.makedirs(salida, exist_ok=True)
srv = subprocess.Popen([sys.executable, "-I", mock, puerto, modo])
time.sleep(1.0)
try:
    html = os.path.join(salida, f"resultado_{tc}{suf}.html")
    consola = os.path.join(salida, f"consola_{tc}{suf}.txt")
    a = [os.path.join(os.path.dirname(__file__), "node_modules/.bin/newman"), "run", col, "-r", "cli,htmlextra",
         "--reporter-htmlextra-export", html, "--reporter-htmlextra-title", f"{tc}{suf}", "--reporter-htmlextra-logs",
         "--reporter-htmlextra-skipHeaders", "Authorization,X-Sandbox-Secret",
         "--reporter-htmlextra-hideRequestBody", "Z00 Login Veterinario,Z50 Login Productor",
         "--reporter-htmlextra-hideResponseBody", "Z00 Login Veterinario,Z01 Perfil Veterinario,Z50 Login Productor",
         "--disable-unicode", "--color", "off",
         "--env-var", f"base_url=http://127.0.0.1:{puerto}", "--env-var", "sandbox_secret=secreto-sandbox-de-prueba-123456",
         "--env-var", f"run_id=qa-m08-g018-prueba-{modo}", "--env-var", "correo_vet=vet@prueba.invalid",
         "--env-var", "contrasena_vet=clave-de-prueba-vet", "--env-var", "correo_productor=prod@prueba.invalid",
         "--env-var", "contrasena_productor=clave-de-prueba-prod", "--env-var", "id_activo_a=885",
         "--env-var", "id_activo_otra_finca=2556", "--env-var", "id_activo_baja=2946"]
    r = subprocess.run(a, capture_output=True, text=True, env={**os.environ, "NO_COLOR": "1", "FORCE_COLOR": "0"})
    open(consola, "w", encoding="utf-8").write(r.stdout + r.stderr)
    lineas = (r.stdout + r.stderr).splitlines()
finally:
    srv.terminate()

cps, trazas, bloque = [], [], None
for l in lineas:
    m = re.match(r"^\s+Pass\s+(CP-\S+ .*)$", l)
    if m: cps.append("OK    | " + m.group(1)); continue
    m = re.match(r"^\s+\d+\.\s+(CP-\S+ .*)$", l)
    if m: cps.append("FALLA | " + m.group(1))

def cerrar(b):
    t = b.strip()
    if t.startswith("'") and t.endswith("'"): t = t[1:-1]
    if re.match(r"^(CP|OBS|PRE|RETIRADO|ABORT|CIERRE)\|", t): trazas.append(t)

for l in lineas:
    m = re.match(r"^\s*\| (.*)$", l)
    if m:
        seg = m.group(1)
        if bloque is not None and seg.startswith("'") and bloque.rstrip().endswith("'"): cerrar(bloque); bloque = None
        bloque = seg if bloque is None else bloque + seg
    elif bloque is not None:
        cerrar(bloque); bloque = None
if bloque is not None: cerrar(bloque)
open(os.path.join(salida, f"checkpoints_{tc}{suf}.txt"), "w", encoding="utf-8", newline="\r\n").write("\n".join(cps) + "\n")
open(os.path.join(salida, f"trazas_{tc}{suf}.txt"), "w", encoding="utf-8", newline="\r\n").write("\n".join(trazas) + "\n")

ESP = ['CP-01', 'CP-02', 'CP-04', 'CP-05', 'CP-06a', 'CP-06b', 'CP-07a', 'CP-07b', 'CP-08', 'CP-09a', 'CP-10', 'CP-11']
ids = [re.match(r"^(OK|FALLA)\s+\|\s+(CP-\S+)", c).group(2) for c in cps]
abort = [t for t in trazas if t.startswith("ABORT|")]
cierre = [t for t in trazas if t.startswith("CIERRE|")]
if abort: estado_runner = "INVALIDA (ABORT) -> exit 3"
elif ids != ESP: estado_runner = f"INVALIDA (ids={ids}) -> exit 3"
else:
    c = subprocess.run(["node", conversor, salida, "1", f"qa-m08-g018-prueba-{modo}"], capture_output=True, text=True)
    estado_runner = f"conversor exit {c.returncode}: {(c.stdout + c.stderr).strip()}"
falla = sum(c.startswith("FALLA") for c in cps)
print(f"[{modo}] newman exit {r.returncode}; checkpoints {len(cps)} (FALLA {falla}); ABORT {len(abort)}; cierre {cierre[:1]}; {estado_runner}")
for c in cps:
    if c.startswith("FALLA"): print("   ", c[:170])
